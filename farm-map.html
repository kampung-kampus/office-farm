function renderPlots(data) {

});

});

}

function selectPlot(p) {

document.querySelectorAll('.plot-item').forEach(el => el.classList.remove('selected'));

const el = document.getElementById(`plot-${p.id}`); if (el) el.classList.add('selected');

map.flyTo([p.lat, p.lng], 19, { duration: 1.2 }); markers[p.id].openPopup();

if (window.innerWidth <= 700) document.getElementById('sidebar').classList.remove('open');

}

function filterPlots() {

const q = document.getElementById('plotSearch').value.toLowerCase();

renderPlots(plots.filter(p => p.plant.toLowerCase().includes(q) || p.plot.toLowerCase().includes(q) || p.type.toLowerCase().includes(q) || p.status.toLowerCase().includes(q)));

}

const legend = document.getElementById('legend');
